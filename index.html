import os, glob import streamlit as st import anthropic from sklearn.feature_extraction.text import TfidfVectorizer from sklearn.metrics.pairwise import cosine_similarity NOM = "Déo IA fiscalité" SYSTEM = f"""Tu es « {NOM} », un assistant spécialisé en fiscalité de la RDC. Tu as été créé par Déo, ton créateur. Si on te demande qui t'a créé, réponds : Déo. Règles : - Réponds en français, de façon claire et précise. - Base-toi UNIQUEMENT sur les extraits de textes fournis et cite le texte/article. - Si l'information n'est pas dans les extraits, dis-le et conseille de vérifier auprès de la DGI ou d'un professionnel. N'invente jamais un taux ou un article. - Tu donnes de l'information, pas un avis juridique officiel.""" @st.cache_resource def charger_textes(): morceaux = [] for f in glob.glob("textes/*.txt"): with open(f, encoding="utf-8") as fh: for p in fh.read().split("\n\n"): if len(p.strip()) > 40: morceaux.append((os.path.basename(f), p.strip())) if not morceaux: return [], None, None vec = TfidfVectorizer() mat = vec.fit_transform([m[1] for m in morceaux]) return morceaux, vec, mat def chercher(question, k=5): morceaux, vec, mat = charger_textes() if not morceaux: return [] scores = cosine_similarity(vec.transform([question]), mat).ravel() idx = scores.argsort()[::-1][:k] return [morceaux[i] for i in idx if scores[i] > 0] st.title(f"🇨🇩 {NOM}") st.caption("Créé par Déo") client = anthropic.Anthropic() # lit la variable ANTHROPIC_API_KEY if "msgs" not in st.session_state: st.session_state.msgs = [] for m in st.session_state.msgs: st.chat_message(m["role"]).write(m["content"]) q = st.chat_input("Pose ta question fiscale...") if q: st.chat_message("user").write(q) extraits = chercher(q) contexte = "\n\n".join(f"[{f}]\n{t}" for f, t in extraits) or "Aucun extrait trouvé." st.session_state.msgs.append({"role": "user", "content": q}) r = client.messages.create( model="claude-sonnet-5-5", max_tokens=1000, system=SYSTEM + "\n\nEXTRAITS :\n" + contexte, messages=st.session_state.msgs, ) rep = r.content[0].text st.session_state.msgs.append({"role": "assistant", "content": rep}) st.chat_message("assistant").write(rep)
